Round 4: built the way a doom theme works. A doom theme sets a handful of slots (keywords, functions, strings, type, constants, numbers, builtin, variables, operators, comments), and doom-themes derives every Emacs face from them: function calls are functions blended 70% into the text color, punctuation is operators at 25%, properties are keywords at 60%, and so on. Card 1 fills the slots the way your current doom-dracula does (strings yellow), card 2 the way doom-one does (strings green). Same laland colors in both. Strings use the yellow you pick below.

# picker.py: font-lock faces exactly as doom-themes computes them
from dataclasses import dataclass
import math

@dataclass(frozen=True)
class Swatch:
    """One named color in OKLCH."""
    name: str
    lc: float = 60.0

def contrast(txt: str, bg: str = "#090e21") -> float:
    t, b = luminance(txt), luminance(bg)
    if abs(b - t) < 0.0005: 
        return 0.0
    return (b ** 0.56 - t ** 0.57) * 114

PINK = Swatch("pink", lc=contrast("#f07fb8"))
print(f"{PINK.name}: {PINK.lc:.0f}")

(defun laland-rebuild ()
  "Rebuild and reload the theme."
  (interactive)
  (shell-command "python3 build.py")
  (load-theme 'doom-laland t))
