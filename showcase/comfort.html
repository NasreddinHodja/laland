Round 9: keywords and types. 1 is the current theme: a saturated sky blue on keywords, and the palest, brightest yellow on every type name, so types out-shout everything. A muted, grown-up mid blue turned out to collide with the pink functions for red-blind readers: to them the pink looks like a greyish periwinkle (#a4a4dd). So the new blues are pale and icy: periwinkle leans violet, steel is grey-blue, cornflower is the bluest. Types move to a calm sea green, so yellow is used only for warnings. A darker gold for types was tried and dropped: color-blind readers couldn't tell it from the orange numbers. Constants go back to a saturated violet, because the pale lavender looked like the icy keywords. Each card runs the color-blind check. Pick a number.

from pathlib import Path
import tomllib

FLOORS = {"code": 60, "comment": 45, "linenr": 30}


class Palette:
    """Load palette.toml and resolve every doom slot to a hex color.

    Slots may reference colors by name or through doom's helpers,
    for example "lighten:magenta:0.4" or "blend:magenta:bg:0.22".
    """

    def __init__(self, path: Path, variant: str = "dark") -> None:
        self.raw = tomllib.loads(path.read_text())[variant]
        self.cache: dict[str, str] = {}

    def get(self, slot: str) -> str:
        if slot in self.cache:
            return self.cache[slot]
        # unknown slots fall back to the foreground, like doom does
        expr = self.raw["slots"].get(slot, "fg")
        kind, *args = expr.split(":")
        match kind:
            case "lighten":
                value = lighten(self.get(args[0]), float(args[1]))
            case "blend":
                a, b, t = args
                value = blend(self.get(a), self.get(b), float(t))
            case _:
                value = self.raw["colors"][kind]
        self.cache[slot] = value
        return value


def report(pal: Palette) -> None:
    for role, floor in FLOORS.items():
        lc = apca(pal.get(role), pal.get("bg"))
        print(f"{role:<10} Lc {lc:5.1f} {'ok' if lc >= floor else 'LOW'}\n")
