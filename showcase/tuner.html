from pathlib import Path
import tomllib

FLOORS = {"code": 60, "comment": 45, "linenr": 30}


class Palette:
    """Load palette.toml and resolve every doom slot to a hex color.

    Slots may reference colors by name or through doom's helpers,
    for example "lighten:magenta:0.4" or "blend:magenta:bg:0.22".
    """

    def __init__(self, path: Path, variant: str = "dark") -> None:
        self.raw = tomllib.loads(path.read_text())[variant]
        self.cache: dict[str, str] = {}

    def get(self, slot: str) -> str:
        if slot in self.cache:
            return self.cache[slot]
        # unknown slots fall back to the foreground, like doom does
        expr = self.raw["slots"].get(slot, "fg")
        kind, *args = expr.split(":")
        match kind:
            case "lighten":
                value = lighten(self.get(args[0]), float(args[1]))
            case "blend":
                a, b, t = args
                value = blend(self.get(a), self.get(b), float(t))
            case _:
                value = self.raw["colors"][kind]
        self.cache[slot] = value
        return value


def report(pal: Palette) -> None:
    for role, floor in FLOORS.items():
        lc = apca(pal.get(role), pal.get("bg"))
        print(f"{role:<10} Lc {lc:5.1f} {'ok' if lc >= floor else 'LOW'}\n")
